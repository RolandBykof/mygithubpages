let isCopying = false;
let fullBookText = "";
let processedFiles = new Set();
let modalElement = null;

document.addEventListener("keydown", async (event) => {
    if (event.key === "F2") {
        isCopying = !isCopying;
        
        if (isCopying) {
            console.log("Kopiointi aloitettu (F3)...");
            fullBookText = "";
            processedFiles.clear();
            if (modalElement) {
                modalElement.close();
                modalElement.remove();
                modalElement = null;
            }
            await startCopying();
        } else {
            console.log("Kopiointi keskeytetty käyttäjän toimesta.");
            isCopying = false;
            showAccessibleModal(fullBookText);
        }
    }
});

async function startCopying() {
    const wait = (ms) => new Promise(resolve => setTimeout(resolve, ms));

    while (isCopying) {
        let currentFile = "";
        let chapterText = "";

        const iframes = document.querySelectorAll('iframe.readium-navigator-iframe');
        
        for (let iframe of iframes) {
            try {
                if (iframe.contentDocument) {
                    const baseEl = iframe.contentDocument.querySelector('base');
                    if (baseEl && baseEl.href) {
                        currentFile = baseEl.href;
                    }
                    if (iframe.contentDocument.body) {
                        chapterText = iframe.contentDocument.body.innerText.trim();
                    }
                }
            } catch (e) {}
        }

        if (currentFile && !processedFiles.has(currentFile)) {
            processedFiles.add(currentFile);
            console.log(`Kopioitu tiedosto: ${currentFile.split('/').pop()}`);
            
            if (chapterText) {
                fullBookText += chapterText + "\n\n--- UUSI LUKU ---\n\n";
            }
        }

        const progressDiv = document.querySelector('div[aria-label="Current progression"]');
        if (progressDiv && progressDiv.innerText.includes("100%")) {
            console.log("Kirja luettu 100%!");
            isCopying = false;
            showAccessibleModal(fullBookText);
            break;
        }

        const forwardButton = document.querySelector('button[aria-label="Go forward"]');
        if (forwardButton) {
            forwardButton.click();
        } else {
            console.log("Seuraava-painiketta ei löytynyt. Lopetetaan.");
            isCopying = false;
            showAccessibleModal(fullBookText);
            break;
        }

        await wait(1500);
    }
}

function showAccessibleModal(text) {
    if (!text) {
        console.log("Ei kopioitavaa tekstiä.");
        return;
    }

    // Luodaan saavutettava dialogi-elementti
    const dialog = document.createElement('dialog');
    dialog.setAttribute('aria-labelledby', 'modal-title');
    dialog.setAttribute('aria-describedby', 'modal-desc');
    dialog.style.padding = '30px';
    dialog.style.borderRadius = '10px';
    dialog.style.border = '3px solid #0056b3';
    dialog.style.boxShadow = '0 8px 30px rgba(0,0,0,0.5)';
    dialog.style.maxWidth = '600px';
    dialog.style.width = '90%';
    dialog.style.backgroundColor = '#ffffff';
    dialog.style.color = '#121212';
    dialog.style.fontSize = '18px';

    dialog.innerHTML = `
        <h2 id="modal-title" style="margin-top:0; color:#0056b3; font-size: 28px;">Kirjan kopiointi valmis</h2>
        <p id="modal-desc" style="line-height: 1.6; margin-bottom: 25px;">
            Kirjan teksti on kopioitu kokonaisuudessaan onnistuneesti. Voit ladata sen koneellesi painamalla alla olevaa latauspainiketta.
        </p>
        <div style="display: flex; gap: 20px; flex-wrap: wrap;">
            <button id="download-btn" style="background-color: #0056b3; color: white; border: 2px solid #003366; padding: 15px 25px; font-size: 18px; font-weight: bold; border-radius: 6px; cursor: pointer;">
                Lataa tiedosto (TXT)
            </button>
            <button id="close-btn" style="background-color: #5a6268; color: white; border: 2px solid #333; padding: 15px 25px; font-size: 18px; border-radius: 6px; cursor: pointer;">
                Sulje ikkuna
            </button>
        </div>
    `;

    document.body.appendChild(dialog);
    modalElement = dialog;

    // Avataan modaali saavutettavasti (showModal lukitsee näkymän ja estää pääsyn taustalle)
    dialog.showModal();

    // Siirretään ruudunlukuohjelman ja käyttäjän fokus heti latauspainikkeeseen
    const downloadBtn = dialog.querySelector('#download-btn');
    const closeBtn = dialog.querySelector('#close-btn');
    downloadBtn.focus();

    const blob = new Blob([text], { type: "text/plain;charset=utf-8" });
    const url = URL.createObjectURL(blob);

    downloadBtn.onclick = () => {
        const a = document.createElement("a");
        a.href = url;
        a.download = "kirja_kokonaisena.txt";
        document.body.appendChild(a);
        a.click();
        document.body.removeChild(a);
        console.log("Tiedosto ladattu onnistuneesti!");
    };

    closeBtn.onclick = () => {
        dialog.close();
    };

    dialog.addEventListener('close', () => {
        dialog.remove();
        modalElement = null;
        console.log("Modaali-ikkuna suljettu.");
    });
}
